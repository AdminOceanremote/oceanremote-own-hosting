<script>
    const CIRCUMFERENCE = 2 * Math.PI * 50; // 314.159...
    const lastUpdateTimes = {};
    const pendingUpdates = {};

    function isValidTemperature(val) {
        // Check if value is a valid number and not empty/error
        if (val === null || val === undefined) return false;
        if (val === '--' || val === '-' || val === '' || val === 0) return false;
        if (isNaN(parseFloat(val))) return false;
        return true;
    }

    function getTempColor(val) {
        if (!isValidTemperature(val)) {
            return "#ef4444"; // Red for error state
        }
        val = parseFloat(val);
        if (val > 8) return "#ef4444";
        if (val >= 0 && val <= 8) return "#10b981";
        if (val > -18 && val < 0) return "#38bdf8";
        return "#0a4d8c";
    }

    function calculatePercentage(val) {
        if (!isValidTemperature(val)) {
            return 0;
        }
        val = parseFloat(val);
        let pct = 0;

        if (val >= 0) {
            if (val <= 8) {
                pct = (val / 8) * 0.7;
            } else {
                const cappedVal = Math.min(val, 20);
                pct = 0.7 + ((cappedVal - 8) / 12) * 0.25;
            }
        } else {
            const cappedVal = Math.max(val, -29);
            pct = (Math.abs(cappedVal) / 29) * 0.95;
        }

        return Math.min(pct, 0.95);
    }

    function updateGauge(id, value, isUpdate = false) {
        const gauge = document.getElementById(`gauge-${id}`);
        const text = document.getElementById(`val-${id}`);
        const pod = document.getElementById(`pod-${id}`);
        const card = document.querySelector(`[onclick="openDiag('${id}')"]`);

        if (!gauge) return;

        const now = Date.now();
        const lastUpdate = lastUpdateTimes[id] || 0;

        if (isUpdate && now - lastUpdate < 300) {
            pendingUpdates[id] = value;
            return;
        }

        const hasValidData = isValidTemperature(value);

        if (!hasValidData) {
            // Error state
            if (card) {
                card.classList.add('error-sensor');
                card.classList.remove('no-data');
            }
            if (text) text.innerText = 'ERROR';

            // Hide the gauge line completely
            gauge.style.stroke = 'transparent';
            gauge.style.strokeDasharray = `0, ${CIRCUMFERENCE}`;
            gauge.classList.remove('glow-active', 'striking');

            lastUpdateTimes[id] = now;
            return;
        }

        // Valid data state
        if (card) {
            card.classList.remove('error-sensor', 'no-data');
            gauge.classList.add('glow-active');
        }

        const val = parseFloat(value);
        const color = getTempColor(val);
        const targetPct = calculatePercentage(val);

        if (isUpdate) {
            // Add pulse animation to pod
            if (pod) {
                pod.classList.remove('pulse-update');
                void pod.offsetWidth;
                pod.classList.add('pulse-update');
            }

            gauge.style.transition = 'stroke-dasharray 0.4s cubic-bezier(0.2, 0.9, 0.3, 1.1), stroke 0.2s ease';
            gauge.classList.add('striking');
            gauge.style.strokeDasharray = `${CIRCUMFERENCE}, 0`;

            setTimeout(() => {
                gauge.style.transition = 'stroke-dasharray 0.5s cubic-bezier(0.3, 0.8, 0.2, 1.1), stroke 0.3s ease';
                gauge.style.strokeDasharray = `${targetPct * CIRCUMFERENCE}, ${CIRCUMFERENCE}`;

                setTimeout(() => {
                    gauge.classList.remove('striking');
                    gauge.style.stroke = color;

                    if (pendingUpdates[id] && Math.abs(parseFloat(pendingUpdates[id]) - val) > 0.01) {
                        const pendingVal = pendingUpdates[id];
                        pendingUpdates[id] = null;
                        setTimeout(() => updateGauge(id, pendingVal, true), 100);
                    }
                }, 150);
            }, 250);
        } else {
            gauge.style.transition = 'stroke-dasharray 0.8s cubic-bezier(0.4, 0, 0.2, 1), stroke 0.4s ease';
            gauge.style.stroke = color;
            gauge.style.strokeDasharray = `${targetPct * CIRCUMFERENCE}, ${CIRCUMFERENCE}`;
        }

        if(text && hasValidData) {
            text.innerText = val.toFixed(2).padStart(5, '0');
        }

        lastUpdateTimes[id] = now;
    }

    function openDiag(id) {
        const meta = document.getElementById(`meta-${id}`);
        const currentVal = document.getElementById(`val-${id}`).innerText;

        document.getElementById('diag-title').innerText = meta.dataset.name;
        document.getElementById('diag-current').innerText = currentVal;
        document.getElementById('diag-min').innerText = meta.dataset.min + '°';
        document.getElementById('diag-max').innerText = meta.dataset.max + '°';
        document.getElementById('diag-avg').innerText = meta.dataset.avg + '°';

        const badge = document.getElementById('diag-health-badge');
        badge.innerText = meta.dataset.health;
        badge.style.color = meta.dataset.hcolor;
        document.getElementById('diag-status-log').innerText = meta.dataset.health;

        document.getElementById('diagOverlay').style.display = 'flex';
        document.body.style.overflow = 'hidden';
        document.getElementById('diagOverlay').dataset.currentDevice = id;
    }

    function closeDiag() {
        document.getElementById('diagOverlay').style.display = 'none';
        document.body.style.overflow = 'auto';
        document.getElementById('diagOverlay').dataset.currentDevice = '';
    }

    async function fetchData() {
        const statusLine = document.getElementById('status-line');
        const syncIcon = document.getElementById('sync-icon');

        try {
            const r = await fetch(window.location.href, {
                headers: {'X-Requested-With': 'XMLHttpRequest'},
                cache: 'no-cache'
            });
            const data = await r.json();

            if (data.devices && Array.isArray(data.devices)) {
                data.devices.forEach((device) => {
                    if (device && device.id !== undefined) {
                        const textEl = document.getElementById(`val-${device.id}`);

                        if (textEl) {
                            const currentDisplayVal = textEl.innerText;
                            const newVal = device.val;

                            const shouldUpdate =
                                (currentDisplayVal === 'ERROR' && isValidTemperature(newVal)) ||
                                (currentDisplayVal !== 'ERROR' && !isValidTemperature(newVal)) ||
                                (currentDisplayVal !== 'ERROR' && isValidTemperature(newVal) &&
                                 Math.abs(parseFloat(currentDisplayVal) - parseFloat(newVal)) > 0.01);

                            if (shouldUpdate) {
                                updateGauge(device.id, newVal, true);

                                const overlay = document.getElementById('diagOverlay');
                                if (overlay.style.display === 'flex' && overlay.dataset.currentDevice === device.id) {
                                    document.getElementById('diag-current').innerText =
                                        isValidTemperature(newVal) ? parseFloat(newVal).toFixed(2).padStart(5, '0') : 'ERROR';
                                }
                            }
                        }
                    }
                });
            }

            // Update status line
            statusLine.classList.add('sync-active');
            if (syncIcon) syncIcon.className = 'fas fa-sync fa-spin me-1';
            statusLine.innerHTML = '<i class="fas fa-sync fa-spin me-1"></i> LIVE_STREAM_ACTIVE';
            setTimeout(() => statusLine.classList.remove('sync-active'), 800);

        } catch (error) {
            console.error("Sync Interrupted:", error);
            statusLine.style.color = "#ef4444";
            statusLine.innerHTML = '<i class="fas fa-exclamation-triangle me-1"></i> CONNECTION_LOST';
        }
    }

    window.addEventListener('DOMContentLoaded', () => {
        document.querySelectorAll('.gauge-circle-fill').forEach((g, i) => {
            const id = g.id.replace('gauge-', '');
            if (g.dataset.raw) {
                setTimeout(() => updateGauge(id, g.dataset.raw, false), i * 150);
            }
        });

        setInterval(fetchData, 2000);
    });
</script>
